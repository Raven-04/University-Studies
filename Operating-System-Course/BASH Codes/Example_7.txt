# Checking the File's Permissions and Changing them

# Check if the file exists using -e.
# Displays the current permissions using ls -l.
# Prompts the user to enter new permissions in numeric format (e.g., 755)
# Uses chmod to apply the new permissions

#!/bin/bash
echo "Enter the file name:"
read file_name

if [ -f "$file_name" ]; then
  echo "File exists. Current permissions:"
  ls -l "$file_name"
  echo "Enter the new permissions in numeric format (e.g., 755):"
  read new_permissions
  chmod "$new_permissions for $file_name updated to:"
  ls -l "$file_name"
else
  echo "File does not exist."
fi
