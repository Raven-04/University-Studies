# Checking the Permissions of a File

# -e evaluates to true if the file or directory specified exists.
# -f for regular files.
# -d for directories.

#!/bin/bash
echo "Enter the file name:"
read file_name

if [ -f "$file_name" ]; then
  echo "File exists: $file_name"
  if [ -r "$file_name" ]; then
    echo "- The file is readable."
  fi
  if [ -w "$file_name" ]; then
    echo "- The file is writable."
  fi
  if [ -x "$file_name" ]; then
    echo "- The file is executable."
  else
    echo "- The file is not executable."
  fi
else
  echo "File does not exist."
fi
